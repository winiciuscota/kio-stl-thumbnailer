cmake_minimum_required(VERSION 3.16)
project(stlthumbnail)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
find_package(KF6 REQUIRED COMPONENTS KIO CoreAddons)
set(CMAKE_AUTOMOC ON)
add_library(stlthumbnail MODULE stlthumbnail.cpp)
target_link_libraries(stlthumbnail Qt6::Gui KF6::KIOGui KF6::CoreAddons)
set_target_properties(stlthumbnail PROPERTIES PREFIX "")
install(TARGETS stlthumbnail DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf6/thumbcreator)
install(PROGRAMS stl-thumbnailer DESTINATION ${KDE_INSTALL_BINDIR})
